
import { useState } from "react";
import "./App.css";

const projects = [
  {
    title: "Nova Studio",
    category: "Web Design",
    description: "A modern and responsive website for a creative digital studio.",
    tech: ["React", "CSS", "Responsive Design"],
  },
  {
    title: "TaskFlow",
    category: "Web Application",
    description: "A clean productivity app for organizing everyday tasks.",
    tech: ["React", "JavaScript", "Local Storage"],
  },
  {
    title: "Weatherly",
    category: "API Project",
    description: "A weather dashboard with a beautiful and intuitive interface.",
    tech: ["React", "API", "CSS"],
  },
];

function App() {
  const [darkMode, setDarkMode] = useState(true);
  const [menuOpen, setMenuOpen] = useState(false);
  const [form, setForm] = useState({
    name: "",
    email: "",
    message: "",
  });
  const [status, setStatus] = useState("");

  function handleSubmit(e) {
    e.preventDefault();

    if (!form.name.trim() || !form.email.trim() || !form.message.trim()) {
      setStatus("Please fill in all fields.");
      return;
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
      setStatus("Please enter a valid email address.");
      return;
    }

    setStatus("Form validated! Connect a form service to receive messages.");
  }

  return (
    <div className={darkMode ? "site dark" : "site light"}>
      <header className="navbar">
        <a className="logo" href="#home">AM<span>.</span></a>

        <button
          className="menu-button"
          onClick={() => setMenuOpen(!menuOpen)}
          aria-label="Toggle navigation"
          aria-expanded={menuOpen}
        >
          ☰
        </button>

        <nav className={menuOpen ? "nav-links open" : "nav-links"}>
          {["Home", "About", "Projects", "Skills", "Contact"].map((item) => (
            <a
              key={item}
              href={`#${item.toLowerCase()}`}
              onClick={() => setMenuOpen(false)}
            >
              {item}
            </a>
          ))}
          <button
            className="theme-button"
            onClick={() => setDarkMode(!darkMode)}
          >
            {darkMode ? "☀ Light" : "☾ Dark"}
          </button>
        </nav>
      </header>

      <main>
        <section className="hero section" id="home">
          <div className="hero-content">
            <p className="eyebrow">CREATIVE DEVELOPER & DESIGNER</p>
            <h1>Hi, I'm <span>Alex Morgan.</span></h1>
            <h2>I build digital experiences that matter.</h2>
            <p className="intro">
              I'm a creative developer and digital designer passionate
              about crafting beautiful, functional and engaging
              experiences for the modern web.
            </p>
            <div className="actions">
              <a href="#projects" className="btn primary">View My Work ↗</a>
              <a href="#contact" className="btn secondary">Get In Touch</a>
            </div>
          </div>
          <div className="hero-art" aria-hidden="true">
            <div className="orb"></div>
            <div className="art-card">DESIGN <span>×</span> CODE</div>
          </div>
        </section>

        <section className="section about" id="about">
          <p className="eyebrow">01 / ABOUT ME</p>
          <h2 className="section-title">A little about <span>me.</span></h2>
          <div className="about-grid">
            <p>
              I'm Alex, a developer and designer who loves bringing
              creative ideas to life through thoughtful design
              and clean code.
            </p>
            <p>
              I focus on building responsive, accessible and
              user-friendly digital products. My goal is to
              create experiences that feel as good as they look.
            </p>
          </div>
        </section>

        <section className="section projects" id="projects">
          <p className="eyebrow">02 / SELECTED WORK</p>
          <h2 className="section-title">Featured <span>projects.</span></h2>
          <div className="project-grid">
            {projects.map((project, index) => (
              <article className="project-card" key={project.title}>
                <div className={`project-image project-${index + 1}`}>
                  <span>{project.title}</span>
                </div>
                <div className="project-body">
                  <p className="eyebrow">{project.category}</p>
                  <h3>{project.title}</h3>
                  <p>{project.description}</p>
                  <div className="tags">
                    {project.tech.map((tech) => (
                      <span key={tech}>{tech}</span>
                    ))}
                  </div>
                </div>
              </article>
            ))}
          </div>
        </section>

        <section className="section skills" id="skills">
          <p className="eyebrow">03 / EXPERTISE</p>
          <h2 className="section-title">My <span>toolkit.</span></h2>
          <div className="skill-grid">
            {[
              "HTML5", "CSS3", "JavaScript", "React",
              "Responsive Design", "UI/UX Design",
              "Figma", "Git & GitHub",
            ].map((skill) => (
              <div className="skill-card" key={skill}>{skill}</div>
            ))}
          </div>
        </section>

        <section className="section contact" id="contact">
          <p className="eyebrow">04 / CONTACT</p>
          <h2 className="section-title">Let's create <span>something great.</span></h2>
          <p className="contact-intro">
            Have a project in mind? I'd love to hear about it.
          </p>
          <form className="contact-form" onSubmit={handleSubmit} noValidate>
            <input
              type="text"
              placeholder="Your name"
              aria-label="Your name"
              value={form.name}
              onChange={(e) => setForm({ ...form, name: e.target.value })}
            />
            <input
              type="email"
              placeholder="Your email"
              aria-label="Your email"
              value={form.email}
              onChange={(e) => setForm({ ...form, email: e.target.value })}
            />
            <textarea
              placeholder="Your message"
              aria-label="Your message"
              rows="5"
              value={form.message}
              onChange={(e) => setForm({ ...form, message: e.target.value })}
            />
            <button className="btn primary" type="submit">
              Send Message ↗
            </button>
            {status && <p className="form-status" role="status">{status}</p>}
          </form>
        </section>
      </main>

      <footer className="footer">
        <a className="logo" href="#home">AM<span>.</span></a>
        <p>© {new Date().getFullYear()} Alex Morgan. All rights reserved.</p>
        <a href="#home">Back to top ↑</a>
      </footer>
    </div>
  );
}

export default App;
